\begin{frame}{Menu 1: method}
Channel $i$, event $e$, high-gain value $v_i(e)$:
\begin{align*}
p_i &= \operatorname{median}_e v_i(e), \quad d_i(e) = v_i(e) - p_i, \quad
\mathrm{CM}(e) = \tfrac{1}{n}\textstyle\sum_i d_i(e),\\
d_i(e) &= b_i\,\mathrm{CM}(e) + r_i(e),\\
\sigma_\mathrm{tot}^2 &= \langle d_i^2\rangle = \sigma_\mathrm{coh}^2 + \sigma_\mathrm{inc}^2,
\quad \sigma_\mathrm{coh}^2 = \langle b_i^2\,\mathrm{CM}^2\rangle, \quad
\sigma_\mathrm{inc}^2 = \langle r_i^2\rangle.
\end{align*}
\begin{itemize}
\item $p_i$ pedestal, $d_i$ deviation, $\mathrm{CM}$ common mode of the $n$ channels of the
  chip read at the same trigger, $b_i$ slope of the channel against it, $r_i$ incoherent part;
  $\langle\,\rangle$ over channels and events.
\item Jumps of the chip ($|\mathrm{CM}| > 6\,s_\mathrm{CM}$) and outliers
  ($|d_i - \mathrm{CM}| > 5\,s_i$) set aside; $s$ the robust width, 1.4826 times the median
  absolute deviation.
\item A \emph{blind channel} takes no common mode ($b_i < 0.75$) and is excluded from $n$.
\end{itemize}
\end{frame}

\begin{frame}{Menu 2: method}
\small
Staircase, channel $i$, threshold $\theta$ (DAC units), rate $R_i(\theta)$:
\begin{align*}
R_i^\mathrm{max} = \max_\theta R_i(\theta), \qquad
\theta_i^\mathrm{e} = \max\{\theta : R_i(\theta) \ge R_i^\mathrm{max}/2\}, \qquad
\Delta\theta = \text{distance between steps of } R_i(\theta).
\end{align*}
Hold-delay scan, shooting channel $i$, delay $\tau$, spectrum $S_i(v;\tau)$, pedestal $p_i$
below breakdown at the same shaping:
\begin{align*}
q_1(\tau) &= \arg\max_{p_i + 48 \le v \le p_i + 400} S_i(v;\tau), &
G &= \operatorname{median}_{\text{last 8 points}} q_1(\tau) - p_i, &
\tau_\mathrm{min} &= \min\{\tau : q_1(\tau) - p_i \ge 0.95\,G\}.
\end{align*}
\begin{itemize}
\item $\theta^\mathrm{e}$ edge of the noise plateau; $\Delta\theta$ one photoelectron in DAC
  units.
\item $q_1$ position of the 1-pe peak: the response of the peak detector to one avalanche,
  read on the peak, not on the mean (which also collects the dark pulses in the window).
\item $G$ single-photoelectron gain; $\tau_\mathrm{min}$ minimum hold delay.
\item Model of the front end: maximum of the shaped pulse 31~ns (25~ns) and 58~ns (50~ns)
  after the crossing of the threshold.
\end{itemize}
\end{frame}
